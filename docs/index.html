快速开始

三行代码接入

比 KCP 更简单的 API。

import "github.com/neko233-com/qcp-lib-go/qcp" // 创建连接 conn := qcp.Dial("game.example.com:9000") // 发送数据 — 零拷贝，自适应 FEC conn.Send([]byte("hello")) // 接收数据 buf := make([]byte, 1024) n, _ := conn.Recv(buf)
using QcpLib; // 创建连接 var conn = new QcpConnection("game.example.com", 9000); // 发送数据 conn.Send(Encoding.UTF8.GetBytes("hello")); // 接收数据 byte[] buf = new byte[1024]; int n = conn.Receive(buf);
#include <qcp/qcp.h> // 创建连接 qcp_conn_t *conn = qcp_dial("game.example.com", 9000); // 发送数据 qcp_send(conn, data, len); // 接收数据 char buf[1024]; int n = qcp_recv(conn, buf, sizeof(buf));
# 运行 QCP vs KCP 对比测试 $ go run . -protocol qcp -duration 30s -connections 1000 $ go run . -protocol kcp -duration 30s -connections 1000 # 输出报告 $ go run . -report html -output results.html # 预期结果: QCP 延迟 -40%, 带宽 -40%
